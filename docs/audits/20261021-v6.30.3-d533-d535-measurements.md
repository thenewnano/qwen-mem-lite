## D#533: the scrubber skipped a BEGIN inside an earlier key's span

`scrubCutOffKeys` set `KEY_BEGIN_RE.lastIndex` to a key's end, so a BEGIN inside that key's span was
never read. With the earlier key at one escape depth and a BEGIN on its header line at another, the
second key's body runs past the first key's end. D#533's input
(`K\nX-hA: K\\nX-hB: v\\n<LF>MII…\\nMII…\\nMII…\n tail`, K = `-----BEGIN RSA PRIVATE KEY-----`) kept
the second and third MII lines (repro.mjs: `L1 scrubbed / L2 STORED / L3 STORED`). The fix hands every
match of `KEY_BEGIN_RE` to `cutOffKeyEnd`; one inside the previous key's span grows the cut to its own
key's end.

## D#535: what v6.30.2's doctor leaves, and which repair keeps every row and a sound file

A fresh subagent ran this on `git archive 00290c19`, with the fixture and plants of the v6.30.3 defects
review (20261021-v6.30.3-pretag-defects.md, Appendix A: 79 WAL pages x 11 variants = 869 plants). Its
results file follows verbatim, then its scripts. One correction to it: its C3 section says `rm -f` of
the WAL drops the 41 observations held there, but doctor prints no such command on those plants; the
claims review traced the drop to v6.30.2's own WAL recovery on open (20261021-v6.30.3-pretag2-claims.md,
F18). What it says for the next attempt at D#535:

- v6.30.2's doctor prints no ✗ line while integrity_check fails afterwards on 114 of 869 plants. They
  are mostly index pages (60) and content tables (50). FTS5 shadow tables are 3: doctor already prints
  `✗ FTS5 rebuild errors` on 37 of the 40 FTS5-shadow plants it leaves failing.
- Of four repairs, copying the readable rows into a fresh file passes all three checks (rows equal
  the record, integrity_check ok, a clean doctor run) on the most plants: 114 of 116 (VACUUM 18, DROP + recreate the FTS
  tables 0, writable_schema delete + VACUUM 59). Traps a copy has to handle: initSchema's own
  `schema_version` row, `foreign_keys = ON`, and a damaged CREATE statement under which every source row
  reads NULL in a column the fresh schema declares NOT NULL (p10-v5): all 241 observation inserts fail,
  and the copy, with no observations, passes integrity_check and doctor reads it as healthy. A copy must
  compare what it wrote with what it read.
- With `writable_schema=ON` on the read-only source, the copy recovers every table in full on 23 of the
  46 plants where doctor prints `✗ Database` (one run; the claims review's re-run read 24, and some
  plants do not read the same twice); today's set-aside remedy leaves 0 rows.
- The copy works with another process holding the store, but that process's close can still checkpoint
  the damaged frame into the old file, so the old file is a fallback only if every session is closed
  first or the file is copied before the repair.

```js
  db.close();

  raw.close();
  return base;

export function cks(buf, s0, s1, be) {
  for (let i = 0; i + 7 < buf.length; i += 8) {
    const a = be ? buf.readUInt32BE(i) : buf.readUInt32LE(i);
    const b = be ? buf.readUInt32BE(i + 4) : buf.readUInt32LE(i + 4);
    s0 = (s0 + a + s1) >>> 0;
    s1 = (s1 + b + s0) >>> 0;
  }
  return [s0, s1];
}
/** pristine main + the WAL with bytes [off, off+len) of the newest frame of `pageOf(db)` filled, chain re-signed, zero -shm. */
export function plant(dir, pageOf, off, len, fill) {
  const base = build();
  const p = fresh(dir);
  writeFileSync(p, readFileSync(join(base, 'pristine.db')));
  writeFileSync(p + '-wal', readFileSync(join(base, 'wal.orig')));
  const ro = new Database(p, { readonly: true });
  const pg = pageOf(ro);
  ro.close();
  writeFileSync(p, readFileSync(join(base, 'pristine.db')));
  rmSync(p + '-shm', { force: true });
  const wal = Buffer.from(readFileSync(join(base, 'wal.orig')));
  const be = wal.readUInt32BE(0) === 0x377f0683;
  const ps = wal.readUInt32BE(8);
  const frames = [];
  for (let o = 32; o + 24 + ps <= wal.length; o += 24 + ps) frames.push({ o, pg: wal.readUInt32BE(o) });
  const t = frames.filter((f) => f.pg === pg).at(-1);
  wal.fill(fill, t.o + 24 + off, t.o + 24 + off + len);
  let [s0, s1] = cks(wal.subarray(0, 24), 0, 0, be);
  for (const f of frames) {
    [s0, s1] = cks(wal.subarray(f.o, f.o + 8), s0, s1, be);
    [s0, s1] = cks(wal.subarray(f.o + 24, f.o + 24 + ps), s0, s1, be);
    wal.writeUInt32BE(s0, f.o + 16);
    wal.writeUInt32BE(s1, f.o + 20);
  }
  writeFileSync(p + '-wal', wal);
  writeFileSync(p + '-shm', Buffer.alloc(32768));
  return pg;
}
export function walPages() {
  const base = build();
  const wal = readFileSync(join(base, 'wal.orig'));
  const ps = wal.readUInt32BE(8);
  const pages = new Set();
  let frames = 0;
  for (let o = 32; o + 24 + ps <= wal.length; o += 24 + ps) {
    pages.add(wal.readUInt32BE(o));
    frames++;
  }
  return { pages: [...pages].sort((a, b) => a - b), frames, ps, walBytes: wal.length };
}

export const SHADOW_RE = /_fts_(data|idx|docsize|config|content)$/;
/** Ordinary tables: type='table', not a virtual table, not an FTS shadow table. */
export function ordinaryTables(db) {
  return db
    .prepare(`SELECT name, sql FROM sqlite_master WHERE type='table' ORDER BY name`)
    .all()
    .filter((r) => !/^CREATE VIRTUAL TABLE/i.test(r.sql || '') && !SHADOW_RE.test(r.name))
    .map((r) => r.name);
}
export function ftsTables(db) {
  return db
    .prepare(`SELECT name FROM sqlite_master WHERE type='table' AND sql LIKE 'CREATE VIRTUAL TABLE%' AND lower(sql) LIKE '%using fts5%' ORDER BY name`)
    .all()
    .map((r) => r.name);
}
const h1 = (s) => createHash('sha1').update(s).digest('hex');
/** count, sha1 over JSON rows ordered by rowid (all columns), and per-row short hashes. */
export function tableDigest(db, t, { rows = false } = {}) {
  try {
    const it = db.prepare(`SELECT * FROM "${t}" ORDER BY rowid`).raw(true).iterate();
    const h = createHash('sha1');
    let n = 0;
    const rh = [];
    for (const r of it) {
      const j = JSON.stringify(r, (_k, v) => (typeof v === 'bigint' ? v.toString() : Buffer.isBuffer(v) ? 'x' + v.toString('hex') : v));
      h.update(j + '\n');
      n++;
      if (rows) rh.push(h1(j).slice(0, 12));
    }
    return { count: n, hash: h.digest('hex').slice(0, 16), ...(rows ? { rowHashes: rh } : {}) };
  } catch (e) {
    return { err: `${e.code || ''} ${e.message}`.trim().slice(0, 160) };
  }
}
/** Copy main (+ wal, + shm if asked) into a fresh dir under WORK/tmp/<tag>; returns the copy path. */
export function copyStore(dir, tag, { wal = true, shm = false } = {}) {
  const t = fresh(join(WORK, 'tmp', tag));
  copyFileSync(join(dir, DB), t);
  if (wal && existsSync(join(dir, DB + '-wal'))) copyFileSync(join(dir, DB + '-wal'), t + '-wal');
  if (shm && existsSync(join(dir, DB + '-shm'))) copyFileSync(join(dir, DB + '-shm'), t + '-shm');
  return t;
}
export const ic = (db) => {
  try {
    const r = db.pragma('integrity_check').map((x) => x.integrity_check);
    return { ok: r.length === 1 && r[0] === 'ok', text: r.join(' ; ').slice(0, 220) };
  } catch (e) {
    return { ok: false, text: 'ERR ' + `${e.code || ''} ${e.message}`.trim().slice(0, 160) };
  }
};
/** Rows of a store: copy main+wal, open read-write on the copy (as Appendix A's readCopy), digest every ordinary table. */
export function readRows(dir, tag, tables, { rows = false } = {}) {
  const t = copyStore(dir, tag, { wal: true });
  let db;
  try {
    db = new Database(t);
  } catch (e) {
    return { openErr: `${e.code || ''} ${e.message}`.trim().slice(0, 160) };
  }
  try {
    let list = tables;
    if (!list) {
      try {
        list = ordinaryTables(db);
      } catch (e) {
        return { openErr: 'schema: ' + `${e.code || ''} ${e.message}`.trim().slice(0, 160) };
      }
    }
    const out = {};
    for (const tb of list) out[tb] = tableDigest(db, tb, { rows });
    return { tables: out };
  } finally {
    try {
      db.close();
    } catch {}
  }
}
/** integrity_check on a copy of main alone, and on main + remaining wal. */
export function integrity(dir, tag) {
  const res = {};
  {
    const t = copyStore(dir, tag + '-mo', { wal: false });
    try {
      const d = new Database(t, { readonly: true });
      res.mainOnly = ic(d);
      d.close();
    } catch (e) {
      res.mainOnly = { ok: false, text: 'OPENERR ' + e.message.slice(0, 120) };
    }
  }
  {
    const hasWal = existsSync(join(dir, DB + '-wal'));
    res.walPresent = hasWal;
    res.walBytes = hasWal ? readFileSync(join(dir, DB + '-wal')).length : 0;
    const t = copyStore(dir, tag + '-mw', { wal: true });
    try {
      const d = new Database(t);
      res.mainWal = ic(d);
      d.close();
    } catch (e) {
      res.mainWal = { ok: false, text: 'OPENERR ' + e.message.slice(0, 120) };
    }
  }
  return res;
}
export function compareRows(rec, post) {
  // rec/post: { tables: {t: {count,hash}|{err}} } or { openErr }
  if (rec.openErr) return { kept: null, why: 'record unreadable: ' + rec.openErr, diff: [] };
  if (post.openErr) return { kept: false, why: 'post unreadable: ' + post.openErr, diff: Object.keys(rec.tables) };
  const diff = [];
  let recErr = false;
  for (const [t, r] of Object.entries(rec.tables)) {
    const p = post.tables[t];
    if (r.err) recErr = true;
    if (!p || p.err || r.err || p.count !== r.count || p.hash !== r.hash) diff.push(t);
  }
  return { kept: diff.length === 0, recErr, diff };
}
```

```js
// Per plant: plant → rows of record (copy, before doctor) → doctor (own process) → post checks.
// Keeps the post-doctor files (main, wal, shm as doctor left them) for Q2/C3 candidates.
import { join } from 'node:path';

import { WORK, DB, plant, readRows, integrity, doctor, dbLines, compareRows, sha } from './common.mjs';

const [slot, kS, KS] = process.argv.slice(2);
const k = Number(kS), K = Number(KS);
const facts = JSON.parse(readFileSync(join(WORK, 'results/facts.json'), 'utf8'));
const ORD = facts.ordinaryTables;
const TRUTH = facts.unplantedRows;
export const VARIANTS = [
  [0, 64, 0xff], [8, 64, 0xff], [100, 64, 0xff], [512, 64, 0xff], [2048, 64, 0xff], [3000, 64, 0xff], [4032, 64, 0xff],
  [100, 8, 0x41], [1500, 8, 0x41], [3000, 8, 0x41], [4000, 8, 0x41],
];

  const recDigest = rec.openErr ? { openErr: rec.openErr } : { tables: Object.fromEntries(Object.entries(rec.tables).map(([t, r]) => [t, r.err ? { err: r.err } : { count: r.count, hash: r.hash }])) };

  if (keepIt) {
    // per-row hashes of the record, for the C3 partial-recovery count
    writeFileSync(join(keep, 'record.json'), JSON.stringify(rec));
  } else {
    rmSync(keep, { recursive: true, force: true });
  }
  row.kept = keepIt;
  appendFileSync(out, JSON.stringify(row) + '\n');
  rmSync(dir, { recursive: true, force: true });
}
```

### repair.mjs
```js
// node repair.mjs <cand> <dir> [newDir]  — one candidate repair on the store in <dir>, in its own process.
// Prints one JSON line: { steps: [...], errors: [...], ... }.
//   C1  VACUUM on a read-write connection.
//   C2  DROP TABLE every FTS5 table (stop on the first throw), initSchema, 'rebuild' every FTS table.
//   C3  new store via initSchema in <newDir>; open <dir> read-only separately; copy every ordinary table's
//       rows (column intersection, rowids kept; forward scan, then a backward scan past a read error);
//       'rebuild' every FTS table in the new file.  C3w: same with writable_schema=ON on the old handle.
//   C4  writable_schema=ON, delete the sqlite_master rows of every FTS5 table and its shadow tables,
//       writable_schema=OFF, VACUUM, then initSchema + 'rebuild'.
import { join } from 'node:path';
import { Database, initSchema, DB, ordinaryTables, ftsTables, SHADOW_RE, sha } from './common.mjs';

const [cand, dir, newDir] = process.argv.slice(2);
const p = join(dir, DB);
const res = { cand, steps: [], errors: [] };
const E = (step, e) => res.errors.push(`${step}: ${e.code || ''} ${e.message}`.trim().slice(0, 200));
const prep = (db) => {
  db.pragma('journal_mode = WAL');
  db.pragma('busy_timeout = 5000');
  db.pragma('foreign_keys = OFF');
};
function rebuildAll(db, tag) {
  let list = [];
  try {
    list = ftsTables(db);
  } catch (e) {
    E(`${tag}: list fts`, e);
  }
  for (const f of list) {
    try {
      db.exec(`INSERT INTO ${f}(${f}) VALUES('rebuild')`);
      res.steps.push(`rebuild ${f}`);
    } catch (e) {
      E(`rebuild ${f}`, e);
    }
  }
}
function main() {
  if (cand === 'C1') {
    let db;
    try {
      db = new Database(p);
    } catch (e) {
      return E('open', e);
    }
    try {
      db.exec('VACUUM');
      res.steps.push('VACUUM');
    } catch (e) {
      E('VACUUM', e);
    } finally {
      try { db.close(); } catch (e) { E('close', e); }
    }
    return;
  }
  if (cand === 'C2') {
    let db;
    try {
      db = new Database(p);
      prep(db);
    } catch (e) {
      return E('open', e);
    }
    try {
      let fts;
      try {
        fts = ftsTables(db);
      } catch (e) {
        return E('list fts', e);
      }
      res.ftsBefore = fts;
      for (const f of fts) {
        try {
          db.exec(`DROP TABLE ${f}`);
          res.steps.push(`drop ${f}`);
        } catch (e) {
          E(`DROP TABLE ${f}`, e);
          res.stopped = true;
          return;
        }
      }
      try {
        initSchema(db);
        res.steps.push('initSchema');
      } catch (e) {
        E('initSchema', e);
        try { if (db.inTransaction) db.exec('ROLLBACK'); } catch {}
        return;
      }
      try { res.ftsAfterInit = ftsTables(db); } catch (e) { E('list fts after', e); }
      rebuildAll(db, 'C2');
    } finally {
      try { db.close(); } catch (e) { E('close', e); }
    }
    return;
  }
  if (cand === 'C4') {
    let db;
    try {
      db = new Database(p);
      prep(db);
      db.unsafeMode(true);
    } catch (e) {
      return E('open', e);
    }
    try {
      let fts;
      try {
        fts = ftsTables(db);
      } catch (e) {
        return E('list fts', e);
      }
      try {
        db.pragma('writable_schema = ON');
        const names = [];
        for (const f of fts) names.push(f, ...['data', 'idx', 'docsize', 'config', 'content'].map((s) => `${f}_${s}`));
        const del = db.prepare(`DELETE FROM sqlite_master WHERE type='table' AND name = ?`);
        let n = 0;
        for (const nm of names) n += del.run(nm).changes;
        db.pragma('writable_schema = OFF');
        res.steps.push(`deleted ${n} sqlite_master rows`);
      } catch (e) {
        E('writable_schema delete', e);
        return;
      }
      // reopen so the connection re-reads the edited schema
      try {
        db.close();
        db = new Database(p);
        prep(db);
        res.steps.push('reopen');
      } catch (e) {
        E('reopen', e);
        return;
      }
      try {
        db.exec('VACUUM');
        res.steps.push('VACUUM');
      } catch (e) {
        E('VACUUM', e);
      }
      try {
        initSchema(db);
        res.steps.push('initSchema');
      } catch (e) {
        E('initSchema', e);
        try { if (db.inTransaction) db.exec('ROLLBACK'); } catch {}
        return;
      }
      rebuildAll(db, 'C4');
    } finally {
      try { db.close(); } catch (e) { E('close', e); }
    }
    return;
  }
  if (cand === 'C3' || cand === 'C3w') {
    const np = join(newDir, DB);
    res.oldBefore = { main: sha(p), wal: sha(p + '-wal') };
    const ndb = new Database(np);
    prep(ndb);
    initSchema(ndb);
    res.steps.push('initSchema(new)');
    // initSchema ends with foreign_keys = ON; the copy inserts tables in name order, so turn it off again
    res.fkAfterInit = ndb.pragma('foreign_keys', { simple: true });
    ndb.pragma('foreign_keys = OFF');
    const newTables = ordinaryTables(ndb);
    // what initSchema wrote by itself
    res.initRows = {};
    for (const t of newTables) {
      const c = ndb.prepare(`SELECT count(*) c FROM "${t}"`).get().c;
      if (c) res.initRows[t] = { count: c, rows: ndb.prepare(`SELECT * FROM "${t}"`).all() };
    }
    let old;
    try {
      old = new Database(p, { readonly: true, fileMustExist: true });
      if (cand === 'C3w') {
        old.unsafeMode(true);
        old.pragma('writable_schema = ON');
      }
    } catch (e) {
      E('open old', e);
      ndb.close();
      return;
    }
    let oldTables = [];
    try {
      oldTables = ordinaryTables(old);
    } catch (e) {
      E('old schema', e);
    }
    res.copied = {};
    ndb.exec('BEGIN');
    // initSchema's own rows collide with the copied ones (schema_version has no key): clear them first.
    for (const t of Object.keys(res.initRows)) if (oldTables.includes(t)) ndb.exec(`DELETE FROM "${t}"`);
    const order = [...oldTables.filter((t) => t !== 'sqlite_sequence'), ...oldTables.filter((t) => t === 'sqlite_sequence')];
    for (const t of order) {
      if (!newTables.includes(t)) {
        res.errors.push(`${t}: not in the new schema, skipped`);
        continue;
      }
      const c = { read: 0, inserted: 0, readErrors: [], insertErrors: 0, firstInsertError: null };
      res.copied[t] = c;
      try {
        if (t === 'sqlite_sequence') ndb.exec('DELETE FROM sqlite_sequence');
        const oc = old.prepare(`PRAGMA table_info("${t}")`).all();
        const nc = new Set(ndb.prepare(`PRAGMA table_info("${t}")`).all().map((x) => x.name));
        const cols = oc.map((x) => x.name).filter((n) => nc.has(n));
        const ipk = oc.filter((x) => x.pk > 0);
        const hasAlias = ipk.length === 1 && /^integer$/i.test(ipk[0].type) && cols.includes(ipk[0].name);
        // rowid always read first (to resume past a read error); inserted explicitly unless a column aliases it
        const sel = `rowid AS __r, ${cols.map((n) => `"${n}"`).join(', ')}`;
        const insCols = [...(hasAlias ? [] : ['rowid']), ...cols.map((n) => `"${n}"`)];
        const ins = ndb.prepare(`INSERT INTO "${t}" (${insCols.join(', ')}) VALUES (${insCols.map(() => '?').join(', ')})`);
        const rid = (r) => r[0];
        const vals = hasAlias ? (r) => r.slice(1) : (r) => r;
        const seen = new Set();
        const take = (r) => {
          const id = rid(r);
          if (seen.has(id)) return;
          seen.add(id);
          c.read++;
          try {
            ins.run(vals(r));
            c.inserted++;
          } catch (e) {
            c.insertErrors++;
            c.firstInsertError ??= `${e.code || ''} ${e.message}`.slice(0, 120);
          }
        };
        let last = null;
        try {
          for (const r of old.prepare(`SELECT ${sel} FROM "${t}" ORDER BY rowid`).raw(true).iterate()) {
            take(r);
            last = rid(r);
          }
        } catch (e) {
          c.readErrors.push(`fwd after rowid ${last}: ${e.code || ''} ${e.message}`.slice(0, 140));
          // backward from the end, down to the last rowid read forward
          try {
            const q = last === null ? `SELECT ${sel} FROM "${t}" ORDER BY rowid DESC` : `SELECT ${sel} FROM "${t}" WHERE rowid > ${Number(last)} ORDER BY rowid DESC`;
            for (const r of old.prepare(q).raw(true).iterate()) take(r);
          } catch (e2) {
            c.readErrors.push(`bwd: ${e2.code || ''} ${e2.message}`.slice(0, 140));
          }
        }
      } catch (e) {
        c.readErrors.push(`setup: ${e.code || ''} ${e.message}`.slice(0, 140));
      }
    }
    ndb.exec('COMMIT');
    try { old.close(); } catch (e) { E('close old', e); }
    rebuildAll(ndb, 'C3');
    ndb.close();
    res.oldAfter = { main: sha(p), wal: sha(p + '-wal') };
    return;
  }
  E('cand', new Error('unknown candidate ' + cand));
}
try {
  main();
} catch (e) {
  E('uncaught', e);
}
console.log(JSON.stringify(res));
```
