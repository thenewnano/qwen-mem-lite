## D#500 equality check

`d500-eq.mjs <beforeDir> <afterDir> <iters> <seed>` runs both scrubbers on the same generated
inputs and counts outputs that differ. Any difference is a defect of D#500, which changes cost only.

## D#504 key starts

32 PEM files from OpenSSL 3.5.5 (`openssl ecparam -name <curve> -genkey -noout` for prime256v1,
secp384r1, secp521r1, secp256k1, secp224r1, secp192k1, prime192v1, brainpoolP256r1/384r1/512r1, SM2,
sect571r1; `openssl pkcs8 -topk8 -nocrypt` of each; `openssl genpkey` for ED25519, ED448, X25519,
X448, RSA, RSA-PSS; `openssl genrsa -traditional`; one encrypted ED25519 PKCS#8), first base64
line cut to 16, 24 and 39 characters under its own BEGIN line: v6.30.0's scrubber kept 45 of the
96 cut lines, the D#504 tree 0. The keys were not kept; the claims reviewer re-ran the question on
199 freshly generated keys (597 cases) and found 0 kept by the release.

## Scripts

### d500-gen.mjs
```js
// Input generator for the D#500 equality fuzz. makeGen(seed) -> () => text.
export function makeGen(seed0) {
  let seed = seed0 >>> 0 || 1;
  const rnd = () => ((seed = (Math.imul(seed, 1103515245) + 12345) >>> 0) / 2 ** 32);
  const pick = (a) => a[Math.floor(rnd() * a.length)];
  const int = (lo, hi) => lo + Math.floor(rnd() * (hi - lo + 1));
  const B64 = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
  const b64 = (k) => Array.from({ length: k }, () => B64[Math.floor(rnd() * 64)]).join('');
  const KINDS = ['RSA PRIVATE KEY', 'EC PRIVATE KEY', 'PRIVATE KEY', 'OPENSSH PRIVATE KEY', 'PGP PRIVATE KEY BLOCK', 'ENCRYPTED PRIVATE KEY'];
  const PREFIX = ['', '', '', '+', '-', ' ', '> ', '"', "'", '`', '<', '</', '</a>', '\\', '\\n', '""', '"</a>', '     1\t', '12:', "x = '", "'+", "+KEY = '''", '<pre>', '"\\n" + "', 'X-a: ', '- ', '>> '];
  // prefixes that are not inert: they can match a line that is blank with no prefix
  const LIVE = ['"', "'", '`', '<', '</', '"</', '</a>', '\\', '\\n', '""', '"</a>', '<pre>', '"\\n" + "'];
  const BLANKISH = ['', '', '', ' ', '\t', '  ', '"', "'", '`', '""', '"\'', '</a>', '</X-a:>', '"</a>', '"\\n"', '\\n"', '"\\n",', '"" +', '</b> ', ' "', '"</X-b:>', '<', '</', '\\', '+', '-', '> ', '"X-a: ', 'X-a: ', '</a', '"\\\\n"', '"\\r"', '</pre>', '"</X-c:>)', '""+'];
  const HEADER = ['Proc-Type: 4,ENCRYPTED', 'DEK-Info: AES-128-CBC,ABCD', 'Version: 1', 'Comment: x', 'X-a: -----BEGIN RSA PRIVATE KEY-----', 'X-b: -----BEGIN EC PRIVATE KEY-----', 'Comment: -----BEGIN RSA PRIVATE KEY-----', '"X-c: "-----BEGIN RSA PRIVATE KEY-----', 'X-d:-----BEGIN PRIVATE KEY-----'];
  function keyLine() {
    const r = rnd();
    if (r < 0.3) return pick(['MII', 'MHcC', 'MHQC', 'b3BlbnNzaC1rZXktdjE', 'lQ', '']) + b64(int(10, 64));
    if (r < 0.45) return b64(int(1, 15));
    if (r < 0.55) return b64(int(16, 50)) + pick(['…', '...', '"', "'", ' [cut]', '-----', '<']);
    if (r < 0.6) return '=' + b64(4);
    if (r < 0.75) return pick(['hello world', 'Done', 'token: alicebob99', '***PEM_KEY***', 'see above']);
    return b64(int(16, 70));
  }
  function group(ls) {
    if (rnd() < 0.3) ls.push(pick(['$ head -c 80 k.pem', 'diff --git a/k b/k', 'Here:', 'prose line']));
    const p = pick(PREFIX);
    const kind = pick(KINDS);
    ls.push(`${p}-----BEGIN ${kind}-----` + (rnd() < 0.08 ? ' ' + b64(int(16, 40)) : ''));
    const lp = () => (rnd() < 0.6 ? p : pick(PREFIX));
    for (let i = int(0, 6); i > 0; i--) {
      const r = rnd();
      if (r < 0.45) ls.push(lp() + pick(HEADER));
      else if (r < 0.75) for (let k = int(1, 25); k > 0; k--) ls.push(rnd() < 0.7 ? pick(BLANKISH) : lp() + pick(BLANKISH));
      else ls.push(lp() + keyLine());
    }
    for (let k = int(0, 40); k > 0; k--) ls.push(rnd() < 0.8 ? pick(BLANKISH) : lp() + pick(BLANKISH));
    for (let i = int(0, 4); i > 0; i--) ls.push(lp() + keyLine());
    if (rnd() < 0.5) ls.push(lp() + `-----END ${kind}-----` + pick(['', "'''", ' more', '"', '\\n"']));
    if (rnd() < 0.3) ls.push(pick(['done', '', 'more prose here']));
  }
  // A BEGIN and header BEGINs under a prefix that is not inert, over a run of blank lines that holds
  // lines this prefix reads as headers or as text, then key material.
  function liveGroup(ls) {
    const p = pick(LIVE);
    ls.push(`${p}-----BEGIN ${pick(KINDS)}-----`);
    for (let i = int(0, 5); i > 0; i--) ls.push(p + pick(['X-a: -----BEGIN RSA PRIVATE KEY-----', 'X-b:-----BEGIN EC PRIVATE KEY-----', 'Comment: -----BEGIN RSA PRIVATE KEY-----', 'Version: 1']));
    for (let k = int(1, 60); k > 0; k--) {
      const r = rnd();
      if (r < 0.55) ls.push(pick(['', '', ' ', '"', '""', '</a>', '"</a>', '\\n"', '"" +']));
      else if (r < 0.8) ls.push(p + pick(['X-b:>', 'X-c:>', 'Version:>', ' ', '', '>', 'n"', '"', '</a>', '+']));
      else ls.push(pick(BLANKISH));
    }
    for (let i = int(0, 4); i > 0; i--) ls.push((rnd() < 0.5 ? p : '') + keyLine());
  }
  function render(ls) {
    const r = rnd();
    if (r < 0.35) return ls.join('\n');
    if (r < 0.45) return ls.join('\r\n');
    if (r < 0.5) return ls.join('\r');
    if (r < 0.65) return JSON.stringify({ content: ls.join('\n') });
    if (r < 0.72) return JSON.stringify(JSON.stringify(ls.join('\n')));
    if (r < 0.82) return ls.map((l, i) => `${String(i + 1).padStart(6)}\t${l}`).join('\n');
    if (r < 0.92) return ls.map((l) => `"${l}\\n" +`).join('\n');
    return ls.map((l) => `'${l}\\n' +`).join('\n') + "\n''";
  }
  // Labelled values after the runs, one scrubbed per pass (the prose guard), each a different
  // length: the text after a run shrinks by a different amount on every pass.
  function tail() {
    const vals = [];
    for (let k = int(2, 12); k > 0; k--) vals.push(`${pick(['secret', 'token'])}: ${'abcdefghijklmnop'.slice(0, int(6, 16))}`);
    return vals.join(' ');
  }
  return () => {
    const ls = [];
    for (let g = int(1, 3); g > 0; g--) (rnd() < 0.45 ? liveGroup : group)(ls);
    const t = render(ls);
    return rnd() < 0.5 ? t + '\n' + tail() : t;
  };
}
```

### d500-eq.mjs
```js
// D#500 equality fuzz: the new scrubber must return byte-identical output to HEAD's on every input.
// argv: oldDir newDir iters seed
const [oldDir, newDir, itersArg, seedArg] = process.argv.slice(2);
const { scrubSecrets: o } = await import(`${oldDir}/secret-scrub.mjs`);
const { scrubSecrets: n } = await import(`${newDir}/secret-scrub.mjs`);
const { makeGen } = await import(new URL('./d500-gen.mjs', import.meta.url));
const gen = makeGen(Number(seedArg));
let diffs = 0;
let scrubbed = 0;
const ex = [];
const iters = Number(itersArg);
for (let i = 0; i < iters; i++) {
  const t = gen();
  const a = o(t);
  const b = n(t);
  if (a !== t) scrubbed++;
  if (a === b) continue;
  diffs++;
  if (ex.length < 3) ex.push({ t, a, b });
}
console.log(JSON.stringify({ iters, seed: Number(seedArg), scrubbedSomething: scrubbed, diffs }));
for (const e of ex) console.log(JSON.stringify(e));
```

### d500-time2.mjs
```js
// Time crafted shapes against benign text, HEAD and the new tree in the same process, interleaved.
const [oldDir, newDir] = process.argv.slice(2);
const { scrubSecrets: o } = await import(`${oldDir}/secret-scrub.mjs`);
const { scrubSecrets: n } = await import(`${newDir}/secret-scrub.mjs`);
const N = 200000;
const CHAIN = Array.from({ length: 40 }, (_, i) => (i % 2 ? 'secret: abcdefgh' : 'token: abcdefgh')).join(' ');
const hdr = (pre) => Array.from({ length: 16 }, (_, i) => `${pre}X-h${String.fromCharCode(65 + i)}: -----BEGIN RSA PRIVATE KEY-----\n`).join('');
const shapes = {
  benign: CHAIN + ' ' + 'the quick brown fox jumps over a lazy dog '.repeat(Math.ceil(N / 42)).slice(0, N),
  d500: CHAIN + ' -----BEGIN RSA PRIVATE KEY-----\n' + hdr('') + '\n'.repeat(N),
  spaces: CHAIN + ' -----BEGIN RSA PRIVATE KEY-----\n' + hdr('') + ' \n'.repeat(N / 2),
  quotes: CHAIN + ' -----BEGIN RSA PRIVATE KEY-----\n' + hdr('') + '"\n'.repeat(N / 2),
  tags: CHAIN + ' -----BEGIN RSA PRIVATE KEY-----\n' + hdr('') + '</a>\n'.repeat(N / 5),
  // headers under a prefix that is not inert, over lines that prefix can match
  liveQuote: CHAIN + '\n"-----BEGIN RSA PRIVATE KEY-----\n' + hdr('"') + '"\n'.repeat(N / 2),
  liveTag: CHAIN + '\n</-----BEGIN RSA PRIVATE KEY-----\n' + Array.from({ length: 16 }, (_, i) => `</X-h${String.fromCharCode(65 + i)}:-----BEGIN RSA PRIVATE KEY-----\n`).join('') + Array.from({ length: N / 9 }, (_, i) => `</t${i % 997}>\n`).join(''),
  escaped: CHAIN + ' -----BEGIN RSA PRIVATE KEY-----\\n' + hdr('').replace(/\n/g, '\\n') + '\\n'.repeat(N / 2),
  headerCap: CHAIN + ' -----BEGIN RSA PRIVATE KEY-----\n' + Array.from({ length: Math.floor(N / 40) }, (_, i) => `X-h${i % 26}: -----BEGIN RSA PRIVATE KEY-----\n`).join(''),
};
const ms = (f, t) => { const s = performance.now(); f(t); return performance.now() - s; };
const best = (f, t) => Math.min(...[0, 1, 2, 3, 4].map(() => ms(f, t)));
for (const t of Object.values(shapes)) { o(t); n(t); if (o(t) !== n(t)) console.log('OUTPUT DIFFERS'); }
const bo = best(o, shapes.benign), bn = best(n, shapes.benign);
console.log(`benign  HEAD ${bo.toFixed(1)} ms  new ${bn.toFixed(1)} ms`);
for (const [k, t] of Object.entries(shapes)) {
  if (k === 'benign') continue;
  const a = best(o, t), b = best(n, t);
  console.log(`${k.padEnd(10)} HEAD ${a.toFixed(1).padStart(7)} ms ${(a / bo).toFixed(1).padStart(5)}x   new ${b.toFixed(1).padStart(7)} ms ${(b / bn).toFixed(1).padStart(5)}x`);
}
```
