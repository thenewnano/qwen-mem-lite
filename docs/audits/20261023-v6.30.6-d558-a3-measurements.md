// node a3/snapcheck.mjs <population> <k> <K> <out>: D#558 attempt 3. For each plant failing the read-only
// check of 602b480e (the round-2 ruler, as tasks/d553/snapcheck3.mjs): this tree's `cli.mjs delete 5
// --confirm` on a copy; for the snapshot it leaves:
//   neu    - this tree's checkSnapshot (a private copy opened by the product opener in a child,
//            rebuildIndexes = the FTS5 rebuilds then REINDEX, integrity_check, observation count);
//   indep  - the system sqlite3, not this tree's code, on a private copy: the four FTS5 'rebuild's,
//            REINDEX, integrity_check, count (the repair arm the second attempt's review asked for);
//   scan   - the system sqlite3's `count(*) FROM observations NOT INDEXED` on a private copy: rows a
//            table scan reads, whatever any index says;
//   file   - the system sqlite3's `PRAGMA integrity_check(1)` on a private copy of the snapshot as it is
//            (the stand-in the withdrawn first attempt used);
//   truth  - the printed restore command run for real in bash in a fresh data dir whose store does not open,
//            then `install.mjs doctor`, then the system sqlite3's integrity_check and an observation count on a
//            copy of what is left;
//   wrote  - whether the listing beside the snapshot changed across the neu check;
//   kind/names - the remedy doctor prints for that directory had the store then stopped opening.
import { spawnSync } from 'node:child_process';
import { appendFileSync, copyFileSync, existsSync, mkdirSync, readdirSync, rmSync, statSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import Database from 'better-sqlite3';
import { DB, WORK, build869, copyPair, plant869, plants869, richPlants, richTemplate } from '../../d546/ruler/plants.mjs';
import { checkIntegrity } from '../db-integrity-602b480e.mjs';
const TREE = new URL('../../../', import.meta.url).pathname;
const { checkSnapshot, dbUnusableRemedy } = await import(`${TREE}/lib/db-unusable.mjs`);
const [pop, kS, KS, out] = process.argv.slice(2);
const k = Number(kS), K = Number(KS), slot = `s8-${pop.replace(/:/g, '_')}-${k}`;
let plants;
if (pop === '869') { build869(); plants = plants869().map((x) => ({ id: x.id, make: (d) => plant869(d, x) })); }
else { const [v, s, c] = pop.split(':'); plants = richPlants(v, Number(c), Number(s), slot).plants; }
const count0 = (p) => { const d = new Database(copyPair(p, join(WORK, 'sc8', slot, 'tpl')), { readonly: true }); try { return d.prepare('SELECT count(*) FROM observations').pluck().get(); } finally { d.close(); } };
const templateObs = pop === '869' ? count0(join(build869(), 'unplanted', DB)) : count0(richTemplate(pop.split(':')[0]));
const home = join(WORK, 'sc8', slot, 'home');
mkdirSync(join(home, 'tmp'), { recursive: true }); mkdirSync(join(home, 'proj'), { recursive: true });
const env = { HOME: home, TMPDIR: join(home, 'tmp'), CLAUDE_PROJECT_DIR: join(home, 'proj'), PATH: process.env.PATH, MEM_NO_AUTO_ADOPT: '1', CLAUDE_MEM_SKIP_UPDATE: '1', CLAUDE_MEM_SKIP_MAINTAIN: '1', CLAUDE_MEM_SKIP_SAVE_ENRICH: '1', MEM_QUIET_HOOKS: '1' };
const count = (p, ro) => { let db; try { db = new Database(p, { readonly: ro, fileMustExist: true }); return db.prepare('SELECT count(*) FROM observations').pluck().get(); } catch (e) { return `ERR ${e.code}`; } finally { try { db?.close(); } catch {} } };
const sqliteIc = (p) => { const r = spawnSync('sqlite3', [p, 'PRAGMA integrity_check(1);'], { encoding: 'utf8', timeout: 120000 }); return r.status === 0 && r.stdout.trim() === 'ok' ? 'ok' : `fail ${(r.stdout + r.stderr).trim().slice(0, 80)}`; };
const HEAL = ['observations_fts', 'session_summaries_fts', 'user_prompts_fts', 'events_fts'].map((t) => `INSERT INTO ${t}(${t}) VALUES('rebuild');`).join('\n') + "\nREINDEX;\nSELECT 'IC>';\nPRAGMA integrity_check(1);\nSELECT '<IC';\nSELECT 'N:' || count(*) FROM observations;\n";
function indep(sp) {
  const c = copyPair(sp, join(WORK, 'sc8', slot, 'indep'));
  const r = spawnSync('sqlite3', ['-batch', c], { input: HEAL, encoding: 'utf8', timeout: 300000 });
  const ic = (r.stdout.match(/^IC>\n([\s\S]*?)\n?<IC$/m) || [])[1]?.trim() ?? null;
  const n = Number((r.stdout.match(/^N:(\d+)$/m) || [])[1] ?? NaN);
  const errors = r.stderr.trim().split('\n').filter(Boolean).slice(0, 3);
  return { ok: errors.length === 0 && ic === 'ok' && n > 0, ic, obs: Number.isNaN(n) ? null : n, errors };
}
function scan(sp) {
  const c = copyPair(sp, join(WORK, 'sc8', slot, 'scan'));
  const r = spawnSync('sqlite3', ['-batch', '-readonly', c, 'SELECT count(*) FROM observations NOT INDEXED;'], { encoding: 'utf8', timeout: 300000 });
  return r.status === 0 ? Number(r.stdout.trim()) : `fail ${r.stderr.trim().slice(0, 60)}`;
}
function truth(sp) {
  // The printed restore command itself, run in a fresh data dir whose store does not open, then doctor
  // ("re-checks"), then the system sqlite3's integrity_check and a count on a copy of what is left. The
  // command is taken with a stub check that calls the snapshot ok, so this arm does not depend on the check.
  const t = join(WORK, 'sc8', slot, 'truth');
  rmSync(t, { recursive: true, force: true }); mkdirSync(t, { recursive: true });
  writeFileSync(join(t, DB), 'GARBAGE not a sqlite file');
  const snap = join(t, `${DB}.pre-delete-x.bak`);
  copyFileSync(sp, snap);
  const withLog = existsSync(`${sp}-wal`) && statSync(`${sp}-wal`).size > 0;
  if (withLog) copyFileSync(`${sp}-wal`, `${snap}-wal`);
  const remedy = dbUnusableRemedy(join(t, DB), new Error('file is not a database'), { check: () => ({ verdict: 'ok', observations: 1, withLog }) });
  const e = { ...env, CLAUDE_MEM_DIR: t };
  const cmd = spawnSync('bash', ['-c', remedy.command], { env: e, cwd: join(home, 'proj'), encoding: 'utf8', timeout: 300000 });
  const doc = spawnSync(process.execPath, [join(TREE, 'install.mjs'), 'doctor'], { env: e, cwd: join(home, 'proj'), encoding: 'utf8', timeout: 300000 });
  const c = copyPair(join(t, DB), join(WORK, 'sc8', slot, 'truthcp'));
  const ic = sqliteIc(c);
  const obs = count(c, true);
  const dbLine = (doc.stdout.split('\n').find((l) => /Database:/.test(l)) || '').trim().slice(0, 80);
  return { cmdRc: cmd.status, ic, obs, dbLine, ok: cmd.status === 0 && ic === 'ok' && typeof obs === 'number' && obs > 0 };
}
for (let i = 0; i < plants.length; i++) {
  if (i % K !== k) continue;
  const pl = plants[i];
  const p = pl.make(join(WORK, 'sc8', slot, 'cur'));
  const ic = checkIntegrity(copyPair(p, join(WORK, 'sc8', slot, 'ic')));
  if (ic.ok || ic.notChecked) continue;
  const storeObs = count(copyPair(p, join(WORK, 'sc8', slot, 'cnt')), true);
  const dir = join(WORK, 'sc8', slot, 'run');
  copyPair(p, dir);
  const del = spawnSync(process.execPath, [join(TREE, 'cli.mjs'), 'delete', '5', '--confirm'], { env: { ...env, CLAUDE_MEM_DIR: dir }, cwd: join(home, 'proj'), encoding: 'utf8', timeout: 60000 });
  const bak = readdirSync(dir).find((n) => n.endsWith('.bak'));
  if (!bak) { appendFileSync(out, JSON.stringify({ id: pl.id, delRc: del.status, bak: null }) + '\n'); continue; }
  const sp = join(dir, bak);
  const before = readdirSync(dir).sort().join(',');
  const t0 = Date.now();
  const neu = checkSnapshot(sp);
  const ms = Date.now() - t0;
  const wrote = readdirSync(dir).sort().join(',') !== before;
  const fcp = join(WORK, 'sc8', slot, 'file.bak'); copyFileSync(sp, fcp);
  const file = sqliteIc(fcp);
  const tr = truth(sp);
  const ind = indep(sp);
  const sc = scan(sp);
  const remedy = dbUnusableRemedy(join(dir, DB), new Error('file is not a database'), { check: checkSnapshot });
  appendFileSync(out, JSON.stringify({ id: pl.id, delRc: del.status, bak: statSync(sp).size, neu: neu.verdict, why: neu.why, reason: neu.reason, neuObs: neu.observations, neuListed: neu.listed ?? null, ms, wrote, file, truth: tr, indep: ind, scan: sc, storeObs, templateObs, kind: remedy.kind, names: remedy.command.includes(sp) }) + '\n');
}

// node a3/sum.mjs <outDir>: summary of a3/snapcheck.mjs's rows, per population and in total.
import { readdirSync, readFileSync } from 'node:fs';
import { join } from 'node:path';
const dir = process.argv[2];
const rows = readdirSync(dir).filter((f) => f.endsWith('.jsonl')).flatMap((f) =>
  readFileSync(join(dir, f), 'utf8').split('\n').filter(Boolean).map((l) => ({ pop: f.replace(/-\d+\.jsonl$/, ''), ...JSON.parse(l) })));
const sum = (rs) => {
  const s = rs.filter((r) => r.bak !== null);
  const c = (f) => s.filter(f).length;
  const tally = (f) => Object.entries(s.reduce((m, r) => { const k = f(r); m[k] = (m[k] || 0) + 1; return m; }, {})).sort((a, b) => b[1] - a[1]);
  const ms = s.map((r) => r.ms).sort((a, b) => a - b);
  return {
    failingPlants: rs.length, snapshots: s.length,
    neu: tally((r) => r.neu + (r.why ? `/${r.why}` : '')),
    truthOk: c((r) => r.truth.ok),
    neuOkTruthOk: c((r) => r.neu === 'ok' && r.truth.ok),
    neuOkTruthNot: s.filter((r) => r.neu === 'ok' && !r.truth.ok).map((r) => `${r.id}:${r.truth.cmdRc}:${r.truth.ic}:${r.truth.obs}`),
    rejectedButRecoverable: s.filter((r) => r.neu !== 'ok' && r.truth.ok).map((r) => `${r.id}:${r.neu}/${r.why || ''}:${r.reason || ''}:obs${r.truth.obs}`),
