```js
// D#525 fixtures: the v6.31.0 pre-tag reviewer's damaged-WAL shape (P1-1), rebuilt.
import { spawnSync } from 'child_process';
import { createHash } from 'crypto';
import { copyFileSync, existsSync, mkdirSync, readFileSync, rmSync, writeFileSync } from 'fs';
import { join } from 'path';
import { homedir } from 'os';

export const REPO = process.env.D525_TREE || '/home/ai/dev/claude-mem-lite';
export const WORK = join(homedir(), '.cache/tmp/d525');
export const DB = 'claude-mem-lite.db';
export const SCRIPTS = new URL('.', import.meta.url).pathname;

export function childEnv(extra = {}) {
  const env = { ...process.env };
  for (const k of Object.keys(env)) {
    if (/API_KEY|OPENROUTER|ANTHROPIC|OPENAI|_PROXY$|_proxy$/i.test(k)) delete env[k];
  }
  return {
    ...env,
    HOME: join(WORK, 'home'),
    TMPDIR: join(WORK, 'tmp'),
    CLAUDE_PROJECT_DIR: join(WORK, 'proj'),
    MEM_NO_AUTO_ADOPT: '1',
    CLAUDE_MEM_SKIP_UPDATE: '1',
    CLAUDE_MEM_SKIP_MAINTAIN: '1',
    CLAUDE_CODE_PATH: join(WORK, 'no-claude'),
    ...extra,
  };
}

export function ensureDirs() {
  for (const d of ['home', 'tmp', 'proj', 'fx']) mkdirSync(join(WORK, d), { recursive: true });
}

export const sha = (p) => (existsSync(p) ? createHash('sha1').update(readFileSync(p)).digest('hex').slice(0, 12) : '-');
export const size = (p) => (existsSync(p) ? readFileSync(p).length : -1);

export function walChecksum(buf, s0, s1, bigEndian) {
  for (let i = 0; i + 7 < buf.length; i += 8) {
    const a = bigEndian ? buf.readUInt32BE(i) : buf.readUInt32LE(i);
    const b = bigEndian ? buf.readUInt32BE(i + 4) : buf.readUInt32LE(i + 4);
    s0 = (s0 + a + s1) >>> 0;
    s1 = (s1 + b + s0) >>> 0;
  }
  return [s0, s1];
}

export function walFrames(wal) {
  const pageSize = wal.readUInt32BE(8);
  const frames = [];
  for (let off = 32; off + 24 + pageSize <= wal.length; off += 24 + pageSize) {
    frames.push({ off, pgno: wal.readUInt32BE(off) });
  }
  return { pageSize, frames };
}

/** Pristine main + original WAL live in WORK/base (made by build.mjs). */
export function plant(dir, pgno, offset, len = 64) {
  rmSync(dir, { recursive: true, force: true });
  mkdirSync(dir, { recursive: true });
  const wal = readFileSync(join(WORK, 'base', 'wal.orig'));
  copyFileSync(join(WORK, 'base', 'pristine.db'), join(dir, DB));
  const bigEndian = wal.readUInt32BE(0) === 0x377f0683;
  const { pageSize, frames } = walFrames(wal);
  const newest = frames.filter((f) => f.pgno === pgno).at(-1);
  if (!newest) throw new Error(`no frame for page ${pgno}`);
  const bad = Buffer.from(wal);
  const start = newest.off + 24 + offset;
  bad.fill(0xff, start, Math.min(start + len, newest.off + 24 + pageSize));
  let [s0, s1] = walChecksum(wal.subarray(0, 24), 0, 0, bigEndian);
  for (const f of frames) {
    [s0, s1] = walChecksum(bad.subarray(f.off, f.off + 8), s0, s1, bigEndian);
    [s0, s1] = walChecksum(bad.subarray(f.off + 24, f.off + 24 + pageSize), s0, s1, bigEndian);
    bad.writeUInt32BE(s0, f.off + 16);
    bad.writeUInt32BE(s1, f.off + 20);
  }
  writeFileSync(join(dir, DB + '-wal'), bad);
  writeFileSync(join(dir, DB + '-shm'), Buffer.alloc(32768));
  return join(dir, DB);
}

export function node(args, env, timeout = 60000) {
  const r = spawnSync(process.execPath, args, { encoding: 'utf8', env, timeout, cwd: SCRIPTS });
  return { code: r.status, out: r.stdout || '', err: r.stderr || '' };
}

/** quick_check of the main file ALONE (copied without its WAL), in a fresh process. */
export function mainOnlyCheck(dbPath) {
  const tmp = join(WORK, 'tmp', `mainonly-${process.pid}-${Date.now()}`);
  mkdirSync(tmp, { recursive: true });
  copyFileSync(dbPath, join(tmp, DB));
  const r = node([join(SCRIPTS, 'qc.mjs'), join(tmp, DB)], childEnv());
  rmSync(tmp, { recursive: true, force: true });
  return r.out.trim() || r.err.trim().split('\n').at(-1);
}
```
