### d550/doctor2.mjs

```javascript
// node doctor2.mjs <k> <K> <outFile>: on each 869 plant, doctor of two trees back to back, each on a
// fresh copy of the plant: MAIN_TREE (6.30.3) and this tree. Per plant: this tree's read-only
// integrity check (lib/db-integrity.mjs) on the store as found. Per arm: whether doctor printed a ✗
// on its Database/FTS5 lines, whether it changed the main file or the log, and the store it left:
// whether it passes that check, and how many truth rows (record(), every ordinary table of the
// unplanted store) it does not hold as they were.
import { spawnSync } from 'node:child_process';
import { appendFileSync, mkdirSync } from 'node:fs';
import { join } from 'node:path';
import { DB, WORK, build869, copyPair, plant869, plants869, record, sha } from '../d546/ruler/plants.mjs';
const TREE = new URL('../../', import.meta.url).pathname;
const MAIN = process.env.MAIN_TREE;
if (!MAIN) throw new Error('MAIN_TREE unset');
const { checkIntegrity } = await import(`${TREE}/lib/db-integrity.mjs`);
const [kS, KS, out] = process.argv.slice(2);
const k = Number(kS);
const K = Number(KS);
const slot = `d2-${k}`;
const sbx = join(WORK, 'sbx', slot);
for (const d of ['home', 'tmp', 'proj']) mkdirSync(join(sbx, d), { recursive: true });
const env = { ...process.env, HOME: join(sbx, 'home'), TMPDIR: join(sbx, 'tmp'), CLAUDE_PROJECT_DIR: join(sbx, 'proj'), MEM_NO_AUTO_ADOPT: '1', CLAUDE_MEM_SKIP_UPDATE: '1', CLAUDE_MEM_SKIP_MAINTAIN: '1', CLAUDE_MEM_SKIP_SAVE_ENRICH: '1', MEM_QUIET_HOOKS: '1' };
for (const key of Object.keys(env))
  if (/API_KEY|OPENROUTER|ANTHROPIC|OPENAI|_PROXY$|^CLAUDECODE$|^CLAUDE_CODE_|^CLAUDE_PID$|^CLAUDE_MEM_DIR$/i.test(key)) delete env[key];
env.PATH = '/usr/bin:/bin';
const truth = record(copyPair(join(build869(), 'unplanted', DB), join(WORK, 'scratch', slot, 'truth')));
const missing = (p) => {
  try {
    const rec = record(p);
    let n = 0;
    for (const [t, m] of Object.entries(truth)) for (const [id, row] of m) if (rec[t]?.get(id) !== row) n++;
    return n;
  } catch (e) {
    return `unreadable: ${e.message.slice(0, 80)}`;
  }
};
const verdict = (r) => (r.ok ? 'ok' : r.notChecked ? `notChecked: ${r.notChecked}` : 'fail');
const plants = plants869();
for (let i = 0; i < plants.length; i++) {
  if (i % K !== k) continue;
  const pl = plants[i];
  const res = { id: pl.id };
  res.asFound = verdict(checkIntegrity(copyPair(plant869(join(WORK, 'doc2', slot, 'af'), pl), join(WORK, 'doc2', slot, 'afc'))));
  for (const [arm, tree] of [['main', MAIN], ['branch', TREE]]) {
    const dir = join(WORK, 'doc2', slot, arm);
    const p = plant869(dir, pl);
    const before = sha(p) + sha(`${p}-wal`);
    const r = spawnSync(process.execPath, [join(tree, 'install.mjs'), 'doctor'], { encoding: 'utf8', env: { ...env, CLAUDE_MEM_DIR: dir }, cwd: join(sbx, 'proj'), timeout: 120000 });
    const lines = r.stdout.split('\n').filter((l) => /Database|FTS5/.test(l) && !/Disk footprint|Deeper/.test(l)).map((l) => l.trim());
    res[arm] = {
      x: lines.some((l) => l.startsWith('✗')),
      changed: before !== sha(p) + sha(`${p}-wal`),
      after: verdict(checkIntegrity(copyPair(p, join(WORK, 'doc2', slot, 'ic')))),
      missing: missing(copyPair(p, join(WORK, 'doc2', slot, 'rc'))),
      lines,
    };
  }
  appendFileSync(out, JSON.stringify(res) + '\n');
}
```

### d550/sum2.mjs

```javascript
// node sum2.mjs <dir>: summary of doctor2.mjs's output. Population F: plants whose read-only check
// fails on the store as found; P: plants it passes; N: not checked.
import { readFileSync, readdirSync } from 'node:fs';
import { join } from 'node:path';
const dir = process.argv[2];
const rows = readdirSync(dir).filter((f) => f.endsWith('.jsonl')).flatMap((f) => readFileSync(join(dir, f), 'utf8').trim().split('\n').filter(Boolean).map((l) => JSON.parse(l)));
const F = rows.filter((r) => r.asFound === 'fail');
const P = rows.filter((r) => r.asFound === 'ok');
const N = rows.filter((r) => r.asFound.startsWith('notChecked'));
const c = (a, f) => a.filter(f).length;
const out = { plants: rows.length, fail: F.length, pass: P.length, notChecked: N.length };
for (const arm of ['main', 'branch']) {
  out[arm] = {
    failX: c(F, (r) => r[arm].x),
    failChanged: c(F, (r) => r[arm].changed),
    failLeftPassingAllRows: c(F, (r) => r[arm].after === 'ok' && r[arm].missing === 0),
    passX: c(P, (r) => r[arm].x),
    passChanged: c(P, (r) => r[arm].changed),
    passLeftFailing: c(P, (r) => r[arm].after !== 'ok'),
    passMissingRows: c(P, (r) => r[arm].missing !== 0),
    integrityLineX: c(F, (r) => r[arm].lines.some((l) => l.startsWith('✗ Database integrity'))),
    databaseLineX: c(F, (r) => r[arm].lines.some((l) => l.startsWith('✗ Database:'))),
  };
}
out.failMainNoX = F.filter((r) => !r.main.x).map((r) => r.id).length;
out.failMainNoXChanged = c(F, (r) => !r.main.x && r.main.changed);
out.failMainFixedIds = F.filter((r) => r.main.after === 'ok' && r.main.missing === 0).map((r) => r.id);
out.notCheckedIds = N.map((r) => `${r.id} ${r.asFound}`);
console.log(JSON.stringify(out, null, 1));
```

### d550/small.mjs

```javascript
// node small.mjs <dir> <variant>: a 300-observation store through this tree's initSchema, WAL mode.
// variant: plain | check (3 rows with a type outside the CHECK list) | notnull (2 rows with a NULL
// project) | unique (a second sdk_sessions row repeating a memory_session_id its UNIQUE index covers) | wal (300
// checkpointed + 50 more left in the log, main and log captured before close).
import Database from 'better-sqlite3';
import { initSchema } from '../../schema.mjs';
import { mkdirSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
const [dir, variant] = process.argv.slice(2);
mkdirSync(dir, { recursive: true });
const p = `${dir}/claude-mem-lite.db`;
const db = new Database(p);
db.pragma('journal_mode = WAL');
initSchema(db);
db.prepare(`INSERT INTO sdk_sessions (content_session_id, memory_session_id, project, started_at, started_at_epoch, status) VALUES ('s1','m1','p0',datetime('now'),1,'active')`).run();
const ins = db.prepare(`INSERT INTO observations (memory_session_id, project, text, type, title, narrative, created_at, created_at_epoch) VALUES ('m1','p0','',?,?,?,datetime('now'),?)`);
db.transaction(() => { for (let i = 0; i < 300; i++) ins.run('discovery', `row ${i}`, `narrative ${i}`, 1.7e12 + i); })();
db.unsafeMode(true);
if (variant === 'check') {
  db.pragma('ignore_check_constraints = 1');
  db.prepare(`UPDATE observations SET type = 'discoverZ' WHERE id IN (5, 6, 7)`).run();
} else if (variant === 'notnull') {
  // NOT NULL taken out of the stored text, the rows written, then put back: the rows break it.
  db.pragma('writable_schema = ON');
  db.prepare(`UPDATE sqlite_master SET sql = replace(sql, 'project TEXT NOT NULL,', 'project TEXT,') WHERE name = 'observations'`).run();
  db.close();
  const d2 = new Database(p);
  d2.prepare(`UPDATE observations SET project = NULL WHERE id IN (10, 11)`).run();
  d2.unsafeMode(true);
  d2.pragma('writable_schema = ON');
  d2.prepare(`UPDATE sqlite_master SET sql = replace(sql, 'project TEXT,', 'project TEXT NOT NULL,') WHERE name = 'observations'`).run();
  d2.pragma('wal_checkpoint(TRUNCATE)');
  d2.close();
  process.exit(0);
} else if (variant === 'unique') {
  // The UNIQUE index made plain in the stored text, a repeated value written, then made UNIQUE
  // again: its b-tree now holds the value twice.
  db.pragma('writable_schema = ON');
  db.prepare(`UPDATE sqlite_master SET sql = replace(sql, 'CREATE UNIQUE INDEX', 'CREATE INDEX') WHERE name = 'idx_sess_memory_sid'`).run();
  db.close();
  const d2 = new Database(p);
  d2.pragma('foreign_keys = OFF'); // session_summaries' key names that column, which needs the index UNIQUE
  d2.prepare(`INSERT INTO sdk_sessions (content_session_id, memory_session_id, project, started_at, started_at_epoch, status) VALUES ('s2','m1','p0',datetime('now'),2,'active')`).run();
  d2.unsafeMode(true);
  d2.pragma('writable_schema = ON');
  d2.prepare(`UPDATE sqlite_master SET sql = replace(sql, 'CREATE INDEX', 'CREATE UNIQUE INDEX') WHERE name = 'idx_sess_memory_sid'`).run();
  d2.pragma('wal_checkpoint(TRUNCATE)');
  d2.close();
  process.exit(0);
} else if (variant === 'wal') {
  db.pragma('wal_checkpoint(TRUNCATE)');
  db.pragma('wal_autocheckpoint = 0');
  db.transaction(() => { for (let i = 0; i < 50; i++) ins.run('discovery', `walrow ${i}`, `in the log ${i}`, 1.8e12 + i); })();
  const main = readFileSync(p);
  const wal = readFileSync(`${p}-wal`);
  db.close();
  writeFileSync(p, main);
  writeFileSync(`${p}-wal`, wal);
  rmSync(`${p}-shm`, { force: true });
  process.exit(0);
}
db.pragma('wal_checkpoint(TRUNCATE)');
db.close();
```

```bash
#!/usr/bin/env bash
# usage: blocks.sh — README.md's "### Damaged database" bash blocks, verbatim, on small stores
# (small.mjs), each in its own sandbox HOME: the constraint rows .recover leaves out, the symbolic-link
# and unreadable-log stops, and the snapshot check and restore (an empty snapshot, a good one, and the
# placeholder pasted as printed).
set -uo pipefail
here=$(cd "$(dirname "$0")" && pwd)
NODE=${NODE:-$(command -v node)}
W="${D546_WORK:?}/blocks"
rm -rf "${W:?}"; mkdir -p "$W"
for v in $(env | grep -oiE '^[A-Z_]*(API_KEY|OPENROUTER|ANTHROPIC|OPENAI)[A-Z_]*' || true); do unset "$v"; done
for v in $(env | grep -oE "^(CLAUDECODE|CLAUDE_CODE_[A-Z_]*)" || true); do unset "$v"; done
unset HTTP_PROXY HTTPS_PROXY http_proxy https_proxy ALL_PROXY all_proxy CLAUDE_MEM_DIR || true
block() { "$NODE" -e '
const s = require("fs").readFileSync(process.argv[1], "utf8");
const sec = s.slice(s.indexOf("### Damaged database"), s.indexOf("## Uninstall"));
process.stdout.write([...sec.matchAll(/```bash\n([\s\S]*?)```/g)].map((m) => m[1])[Number(process.argv[2]) - 1]);' "$here/../../README.md" "$1"; }
run() { (cd "$1" && env -i HOME="$1" PATH=/usr/bin:/bin bash -c "$2"); }
q() { sqlite3 -readonly "$1" "$2" 2>&1 | tr '\n' ' '; }
B1=$(block 1); B2=$(block 2); B3=$(block 3); B4=$(block 4)
for v in check notnull unique; do
  H="$W/$v"; mkdir -p "$H"; "$NODE" "$here/small.mjs" "$H/.claude-mem-lite" "$v"
  D="$H/.claude-mem-lite/claude-mem-lite.db"
  echo "== $v: as found obs=$(q "$D" 'SELECT count(*) FROM observations') sess=$(q "$D" 'SELECT count(*) FROM sdk_sessions') rw-check=$(cp "$D" "$H/c.db"; sqlite3 "$H/c.db" 'PRAGMA integrity_check' 2>&1 | head -2 | tr '\n' ' ')"
  out=$(run "$H" "$B1" 2>&1); echo "   block 1 rc=$? out=[$out]"
  R="$H/cml-recover/recovered.db"
  echo "   recovered obs=$(q "$R" 'SELECT count(*) FROM observations') sess=$(q "$R" 'SELECT count(*) FROM sdk_sessions') lost_and_found=$(q "$R" 'SELECT count(*) FROM lost_and_found')"
done
# Symbolic link: block 1 and block 2 stop at their first step; nothing is created or removed.
H="$W/symlink"; mkdir -p "$H/disk" "$H/.claude-mem-lite"; "$NODE" "$here/small.mjs" "$H/disk" wal
ln -s "$H/disk/claude-mem-lite.db" "$H/.claude-mem-lite/claude-mem-lite.db"
out=$(run "$H" "$B1" 2>&1); echo "== symlink: block 1 rc=$? out=[$out] cml-recover=$([ -e "$H/cml-recover" ] && echo yes || echo no)"
out=$(run "$H" "$B2" 2>&1); echo "   block 2 rc=$? out=[$out] disk=[$(ls "$H/disk" | tr '\n' ' ')]"
# A log that exists but cannot be read: block 1 stops at the copy.
H="$W/unreadable-log"; mkdir -p "$H"; "$NODE" "$here/small.mjs" "$H/.claude-mem-lite" wal
chmod 000 "$H/.claude-mem-lite/claude-mem-lite.db-wal"
out=$(run "$H" "$B1" 2>&1); echo "== unreadable log: block 1 rc=$? out=[$out] recovered.sql=$([ -e "$H/cml-recover/recovered.sql" ] && echo yes || echo no)"
chmod 600 "$H/.claude-mem-lite/claude-mem-lite.db-wal"
# Snapshots: an empty one, a good one, and the placeholder as printed.
H="$W/snap"; mkdir -p "$H"; "$NODE" "$here/small.mjs" "$H/.claude-mem-lite" wal
D="$H/.claude-mem-lite/claude-mem-lite.db"
"$NODE" "$here/small.mjs" "$H/good" plain; cp "$H/good/claude-mem-lite.db" "$D.pre-maintain-2026-10-01T00-00-00-000Z-1-0.bak"
: > "$D.pre-delete-2026-10-22T00-00-00-000Z-1-0.bak"
for s in "$D.pre-delete-2026-10-22T00-00-00-000Z-1-0.bak" "$D.pre-maintain-2026-10-01T00-00-00-000Z-1-0.bak"; do
  c=$(run "$H" "${B3//<snapshot>/$s}" 2>&1); echo "== snapshot $(basename "$s") ($(stat -c %s "$s") bytes): check rc=$? out=[$(echo "$c" | tr '\n' ' ')]"
done
out=$(run "$H" "$B3" 2>&1); echo "== placeholder as printed: check rc=$? out=[$out]"
out=$(run "$H" "$B4" 2>&1); echo "   restore rc=$? out=[$out] wal=$([ -e "$D-wal" ] && echo kept || echo GONE) obs=$(q "$D" 'SELECT count(*) FROM observations')"
s="$D.pre-delete-2026-10-22T00-00-00-000Z-1-0.bak"
out=$(run "$H" "${B4//<snapshot>/$s}" 2>&1); echo "   restore the empty one rc=$? out=[$out] wal=$([ -e "$D-wal" ] && echo kept || echo GONE) obs=$(q "$D" 'SELECT count(*) FROM observations')"
s="$D.pre-maintain-2026-10-01T00-00-00-000Z-1-0.bak"
out=$(run "$H" "${B4//<snapshot>/$s}" 2>&1); echo "   restore the good one rc=$? out=[$out] wal=$([ -e "$D-wal" ] && echo kept || echo GONE) obs=$(q "$D" 'SELECT count(*) FROM observations') check=$(q "$D" 'PRAGMA integrity_check')"
rm -rf "${W:?}"
```

```javascript
// node nextopen.mjs <pg> <v>: one 869 plant; observations SQLite reads from it (read-only, on a copy),
// then one `cli.mjs search` of this tree on it (sandboxed), then the same count and whether the log is left.
import { spawnSync } from 'node:child_process';
import { existsSync, mkdirSync } from 'node:fs';
import { join } from 'node:path';
import Database from 'better-sqlite3';
import { WORK, copyPair, plant869 } from '../d546/ruler/plants.mjs';
const TREE = new URL('../../', import.meta.url).pathname;
const [pg, v] = process.argv.slice(2).map(Number);
const sbx = join(WORK, 'nextopen');
for (const d of ['home', 'tmp', 'proj']) mkdirSync(join(sbx, d), { recursive: true });
const dir = join(sbx, 'store');
const p = plant869(dir, { pg, v });
const count = () => {
  const db = new Database(copyPair(p, join(sbx, 'c')), { readonly: true });
  try {
    return db.prepare('SELECT count(*) FROM observations').pluck().get();
  } catch (e) {
    return e.message;
  } finally {
    db.close();
  }
};
const before = count();
const env = { HOME: join(sbx, 'home'), TMPDIR: join(sbx, 'tmp'), CLAUDE_PROJECT_DIR: join(sbx, 'proj'), CLAUDE_MEM_DIR: dir, PATH: '/usr/bin:/bin', MEM_NO_AUTO_ADOPT: '1', CLAUDE_MEM_SKIP_UPDATE: '1', CLAUDE_MEM_SKIP_MAINTAIN: '1', CLAUDE_MEM_SKIP_SAVE_ENRICH: '1', MEM_QUIET_HOOKS: '1' };
const r = spawnSync(process.execPath, [join(TREE, 'cli.mjs'), 'search', 'alpha'], { env, cwd: join(sbx, 'proj'), encoding: 'utf8', timeout: 120000 });
console.log(JSON.stringify({ plant: `p${pg}-v${v}`, before, after: count(), logLeft: existsSync(`${p}-wal`), searchRc: r.status }));
```

```bash
#!/usr/bin/env bash
# usage: time.sh <store dir> <rounds> — doctor on main and on this tree, alternating, wall ms each.
set -uo pipefail
B=$1; N=${2:-5}
S=$(mktemp -d "$HOME/.cache/tmp/d550-time-XXXX")
export PATH="$HOME/.cache/tmp/d550/bin:/usr/bin:/bin"
for v in $(env | cut -d= -f1 | grep -iE 'API_KEY|OPENROUTER|ANTHROPIC|OPENAI|_PROXY$'); do unset "$v"; done
for i in $(seq 1 "$N"); do
  for tree in "$HOME/.cache/d550-main" /home/ai/dev/claude-mem-lite; do
    t0=$(date +%s%N)
    HOME="$S" TMPDIR="$S" CLAUDE_MEM_DIR="$B" CLAUDE_PROJECT_DIR="$S" CLAUDE_MEM_SKIP_UPDATE=1 CLAUDE_MEM_SKIP_MAINTAIN=1 MEM_NO_AUTO_ADOPT=1 MEM_QUIET_HOOKS=1 node "$tree/install.mjs" doctor > "$S/out.txt" 2>&1
    t1=$(date +%s%N)
    echo "$(basename "$tree") $(( (t1 - t0) / 1000000 )) $(grep -c 'Database integrity: ok' "$S/out.txt") load $(cut -d' ' -f1 /proc/loadavg)"
  done
done
rm -rf "${S:?}"
```

### d550/summary.mjs

```javascript
// node summary.mjs <tag...>: the README recovery's outcome per population.
import { readFileSync, readdirSync } from 'node:fs';
import { join } from 'node:path';
const W = process.env.D546_WORK || `${process.env.HOME}/.cache/tmp/d550`;
const SKIP = new Set(['sqlite_sequence', 'zz_values', 'zz_noindex']);
for (const tag of process.argv.slice(2)) {
  const rows = readdirSync(join(W, 'out', tag))
    .filter((f) => f.endsWith('.jsonl'))
    .flatMap((f) => readFileSync(join(W, 'out', tag, f), 'utf8').trim().split('\n').filter(Boolean).map((l) => JSON.parse(l)));
  const failing = rows.filter((r) => r.integrity === 'fail');
  const stopped = failing.filter((r) => r.rc2 === undefined);
  const placed = failing.filter((r) => r.rc2 === 0);
  const why = {};
  for (const r of stopped) {
    const k = (r.stderr1.split('\n').find((l) => /rror/.test(l)) ?? (r.recoveredCheck || `rc ${r.rc1}`)).replace(/\(\w+\)|near line \d+|main\.\w+/g, '').replace(/"[^"]*"/g, '"…"').slice(0, 70);
    why[k] = (why[k] ?? 0) + 1;
  }
  let missedPlants = 0, missedRows = 0, readIntact = 0, intact = 0, changed = 0, extra = 0, valuesChanged = 0;
  const missedBy = {};
  for (const r of placed) {
    let m = 0;
    for (const [t, v] of Object.entries(r.tables ?? {})) {
      if (t === 'zz_values') valuesChanged += v.changed;
      if (SKIP.has(t)) continue;
      m += v.missedReadable; readIntact += v.readIntact; intact += v.intact; changed += v.changed; extra += v.extra;
      if (v.missedReadable) missedBy[t] = (missedBy[t] ?? 0) + v.missedReadable;
    }
    if (m) { missedPlants++; missedRows += m; }
  }
  const green = placed.filter((r) => r.doctor2?.some((l) => /✓ Database integrity: ok/.test(l)) && !r.doctor2.some((l) => /^✗/.test(l))).length;
  const ftsOk2 = placed.filter((r) => r.doctor2?.some((l) => /FTS5 integrity: all indexes healthy/.test(l))).length;
  const saveOk = placed.filter((r) => r.save?.rc === 0 && r.save.found).length;
  const copyBad = failing.filter((r) => !r.copyAsFound).length;
  console.log(`\n== ${tag}: ${rows.length} plants, ${failing.length} fail the check`);
  console.log(JSON.stringify({ stopped: stopped.length, placed: placed.length, copyNotAsFound: copyBad, doctorGreenAfter: green, ftsHealthyAfter: ftsOk2, saveAndSearchOk: saveOk, missedReadablePlants: missedPlants, missedReadableRows: missedRows, readIntactRows: readIntact, intactRows: intact, changedRows: changed, extraRows: extra, zzValuesChanged: valuesChanged }));
  const byStep = {};
  for (const r of stopped) {
    // `> recovered.sql` creates the file before sqlite3 runs, so the step is read from the message:
    // `.recover` reports "sql error: …", the sqlite3 commands after it "Error: …".
    const k = r.step === 'copy' ? 'copy' : /^sql error:/m.test(r.stderr1) ? '.recover' : /^(Error|Parse error|Runtime error)/m.test(r.stderr1) ? 'after .recover' : `other (rc ${r.rc1})`;
    byStep[k] = (byStep[k] ?? 0) + 1;
  }
  console.log(' stopped at:', JSON.stringify(byStep));
  console.log(' stopped because:', JSON.stringify(why));
  console.log(' missed readable rows by table:', JSON.stringify(missedBy));
  const ms = placed.map((r) => r.ms).sort((a, b) => a - b);
  console.log(` procedure ms median ${ms[Math.floor(ms.length / 2)]} max ${ms.at(-1)}`);
}
```
